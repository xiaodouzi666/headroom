class Latest(v.SupplementEvidence):
    def __init__(self):
        self.root=ROOT
        self.document={**prev.document,"summary":{**prev.document["summary"],"latest_campaign":"R15"}}
        self.levels=copy.deepcopy(prev.levels)
    def selected(self,scenario):
        if scenario=="s3_eval_no_grad_missing":return {"scenario_id":scenario,"campaign":"R15"}
        if scenario=="s8_two_models_resident":return {"scenario_id":scenario,"campaign":"R13"}
        return prev.selected(scenario)
    def value(self,name):
        mapping={"R15.s3_eval_no_grad_missing.before.GB":"selected.s3.before_GB","R15.s3_eval_no_grad_missing.after.GB":"selected.s3.after_GB","best.s3_eval_no_grad_missing.reduction_pct":"selected.s3.reduction_pct"}
        if name in mapping:return claims[mapping[name]]
        if name in previous_claims:return previous_claims[name]
        return prev.value(name)
