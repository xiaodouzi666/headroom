CACHE = {}


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def load(relative):
    if relative not in CACHE:
        raw = (ROOT / relative).read_bytes()
        CACHE[relative] = (json.loads(raw), digest(raw))
    return CACHE[relative]


def pointer(value, path):
    for part in path.split("/")[1:]:
        part = part.replace("~1", "/").replace("~0", "~")
        value = value[int(part)] if isinstance(value, list) else value[part]
    return value


def write(relative, text):
    path = OUT / relative
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x") as handle:
        handle.write(text)
    return {"path": str(path.relative_to(ROOT)), "sha256": digest(path.read_bytes())}


def write_json(relative, value):
    return write(relative, json.dumps(value, indent=2, allow_nan=False) + "\n")


def replace_once(text, old, new):
    assert text.count(old) == 1, (old[:90], text.count(old))
    return text.replace(old, new, 1)


def replace_line(text, needle, transform):
    lines = text.splitlines(keepends=True)
    selected = [index for index, line in enumerate(lines) if needle in line]
    assert len(selected) == 1, (needle, len(selected))
    index = selected[0]
    lines[index] = transform(lines[index])
    return "".join(lines)
