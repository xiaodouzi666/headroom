"""Read-only checks of the R18 preparation draft. Never calls prepare or dispatch."""

import ast
import copy
import hashlib
import json
import runpy
import socket
import subprocess
import sys
from pathlib import Path

sys.dont_write_bytecode = True
ROOT = Path(__file__).resolve().parents[2]
SCRIPT = ROOT / "reports/repair-followup-20261030/prepare-r18.py.txt"
REPORT = ROOT / "reports/repair-followup-20261030/r18-prepare-script-check.json"
checks = []


def check(condition, label):
    if not condition:
        raise AssertionError(label)
    checks.append(label)


def reject(function, label, *args):
    try:
        function(*args)
    except (ValueError, RuntimeError):
        checks.append(label)
    else:
        raise AssertionError(label)


def no_network(*args, **kwargs):
    raise RuntimeError("Read-only preparation review cannot use the network")


def main():
    socket.getaddrinfo = no_network
    socket.socket.connect = no_network
    socket.socket.connect_ex = no_network
    raw = SCRIPT.read_bytes()
    tree = ast.parse(raw, filename=str(SCRIPT))
    compile(tree, str(SCRIPT), "exec")
    module = runpy.run_path(str(SCRIPT), run_name="_r18_preparation_readonly_review")
    check(
        "prepare" in module and "main" in module, "draft parsed without calling either entrypoint"
    )
    outputs = (
        module["PREPARATION"].parent,
        module["PROPOSAL"],
        module["PREFLIGHT"],
        module["EXECUTION"],
    )
    check(
        all(not path.exists() for path in outputs), "all real R18 output paths absent before review"
    )
    check(module["BASE_PROPOSAL"].name == "r14-proposal.json", "R14 remains the proposal anchor")
    check(
        module["BASE_PROPOSAL_SHA"]
        == "f3a8041e026e535a338fa283959929b7bc6f31ba181b1844719d66ff2b734026",
        "R14 proposal SHA unchanged",
    )
    check(
        module["BASE_PREPARATION_SHA"]
        == "a2c1fcf6fbc77ba196c2b559c16f99ed19a6346d902608f76bce43516ba49898",
        "R14 preparation SHA unchanged",
    )
    check(
        module["CONTROLLER_SHA"]
        == "590189eb3b23f1f932c62a849afd1aee26edb9968822964e65a8c3b7f04ed426",
        "controller v2 unchanged",
    )
    check(
        module["FUNDING_SHA"] == "3b5f9e2573fda21f34c4858a4af64b27a5952e5c1c6bb944dcdb3d01a67bf889",
        "same v3 funding receipt",
    )
    check(
        len(module["NEW_REVIEWED_SOURCE"]) == 27,
        "twenty-seven draft implementation identities",
    )
    for name, sha in module["NEW_REVIEWED_SOURCE"].items():
        check(
            hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == sha, "frozen source " + name
        )

    controller = module["load_controller"]()
    base_proposal = module["repair"].read_json(module["BASE_PROPOSAL"])
    check(
        module["repair"].digest(base_proposal) == module["BASE_PROPOSAL_SHA"],
        "R14 proposal canonical digest valid",
    )
    receipt = controller.funding(base_proposal["funding_receipt_path"], module["FUNDING_SHA"])
    state, account = controller.accounting(receipt)
    controller.require_admission(state, receipt)
    module["check_starting_accounting"](account)
    checks.append(
        "read-only real donor holds/profile/admission/accounting match locked starting state"
    )
    expected_account = module["EXPECTED_STARTING_ACCOUNTING"]
    for key in expected_account:
        broken = copy.deepcopy(expected_account)
        broken[key] = "__changed__"
        reject(
            module["check_starting_accounting"],
            "starting accounting changed " + key + " rejected",
            broken,
        )
        del broken[key]
        reject(
            module["check_starting_accounting"],
            "missing starting accounting " + key + " rejected",
            broken,
        )

    base = module["repair"].read_json(Path(base_proposal["prepared_path"]))
    fresh = copy.deepcopy(base)
    fresh["implementation_pins"]["synthetic-review-only.py"] = {"sha256": "a" * 64}
    module["immutable_preparation"](base, fresh)
    checks.append("only new implementation pins accepted by semantic comparison")
    reject(
        module["immutable_preparation"], "unchanged old implementation pack rejected", base, base
    )
    for field in module["SEMANTIC_FIELDS"]:
        broken = copy.deepcopy(fresh)
        broken[field] = "__changed__"
        reject(
            module["immutable_preparation"],
            "changed semantic anchor " + field + " rejected",
            base,
            broken,
        )
    broken = copy.deepcopy(fresh)
    broken["unexpected"] = "mutation"
    reject(
        module["immutable_preparation"], "changed non-implementation field rejected", base, broken
    )
    for field, bad in (("fixed_denominator", 8), ("minimum_targets", 5)):
        b, f = copy.deepcopy(base), copy.deepcopy(fresh)
        b["controls"][field] = f["controls"][field] = bad
        reject(module["immutable_preparation"], "changed " + field + " rejected", b, f)

    requests = base_proposal["requests"]
    module["fixed_requests"](requests, requests)
    checks.append("all nine original R14 bound requests accepted unchanged")
    for key, value in (
        ("budget_usd", "0.61"),
        ("max_rounds", 7),
        ("max_steps", 21),
        ("seed", 1),
        ("max_seconds", 121),
        ("search_mode", "first_success"),
    ):
        broken = copy.deepcopy(requests)
        broken[next(iter(broken))][key] = value
        reject(
            module["fixed_requests"],
            "changed request field " + key + " rejected even against matching fake prior",
            broken,
            broken,
        )
        reject(
            module["fixed_requests"],
            "request " + key + " differs from R14 rejected",
            broken,
            requests,
        )
    shorter = dict(list(requests.items())[:-1])
    reject(module["fixed_requests"], "non-nine request set rejected", shorter, shorter)

    # Exercise only the revision gate with a synthetic git reader: no writes and
    # no invocation of the real draft prepare function.
    gate = module["check_revision"]
    namespace = gate.__globals__
    original_git = namespace["git"]
    sha = "a" * 40
    pins = {name: {"sha256": digest} for name, digest in module["NEW_REVIEWED_SOURCE"].items()}

    def git_ok(*args):
        if args[:2] == ("rev-parse", "HEAD"):
            return sha
        if args[:3] == ("ls-tree", "-r", "--name-only"):
            return "\n".join(pins)
        if args[:2] == ("diff", "--quiet"):
            return ""
        raise AssertionError(args)

    try:
        namespace["git"] = git_ok
        gate(sha, pins)
        checks.append("exact new tracked-clean commit gate accepts matching synthetic source")
        reject(gate, "non-exact revision rejected", "main", pins)
        reject(gate, "different HEAD rejected", "b" * 40, pins)
        extra = dict(pins, **{"untracked.py": {"sha256": "a" * 64}})
        reject(gate, "untracked implementation rejected", sha, extra)
        bad = copy.deepcopy(pins)
        bad[next(iter(bad))]["sha256"] = "b" * 64
        reject(gate, "different reviewed implementation hash rejected", sha, bad)

        def git_dirty(*args):
            if args[:2] == ("diff", "--quiet"):
                raise ValueError("READ_ONLY_GIT_CHECK_FAILED")
            return git_ok(*args)

        namespace["git"] = git_dirty
        reject(gate, "dirty implementation rejected", sha, pins)
    finally:
        namespace["git"] = original_git
    check(
        all(not path.exists() for path in outputs),
        "all real R18 output paths still absent after review",
    )
    head = subprocess.run(
        ["git", "--no-optional-locks", "rev-parse", "HEAD"],
        check=True,
        text=True,
        capture_output=True,
    ).stdout.strip()
    document = {
        "scope": "R18_PREPARATION_DRAFT_READ_ONLY_CHECK_NOT_EXECUTED",
        "script_sha256": hashlib.sha256(raw).hexdigest(),
        "current_head_at_review": head,
        "new_root_commit_still_required": True,
        "source_map_review_state": "provisional until root checkpoint/callsite final independent hashes",
        "write_reviewed_prepare_called": False,
        "real_R18_outputs_created": False,
        "model_calls": 0,
        "GPU_jobs": 0,
        "funding_changed": False,
        "controller_sha256": module["CONTROLLER_SHA"],
        "funding_receipt_sha256": module["FUNDING_SHA"],
        "starting_accounting": {key: account[key] for key in expected_account},
        "source_hashes": module["NEW_REVIEWED_SOURCE"],
        "checks": checks,
        "check_count": len(checks),
        "status": "PASS_PENDING_ROOT_REVIEW_AND_COMMIT",
    }
    REPORT.write_text(json.dumps(document, indent=2) + "\n")
    print(
        json.dumps(
            {
                "status": document["status"],
                "checks": len(checks),
                "script_sha256": document["script_sha256"],
                "report_sha256": hashlib.sha256(REPORT.read_bytes()).hexdigest(),
            }
        )
    )


if __name__ == "__main__":
    main()
