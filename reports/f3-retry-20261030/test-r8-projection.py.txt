def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value))
    return path

def test_partial_fixed_nine_actual_unknown_and_three_distinct_diagnosis_counts(campaign, tmp_path):
    original = {str(p): p.read_bytes() for p in campaign.parent.rglob("*") if p.is_file()}
    summary = projection.stage_campaign(campaign, tmp_path / "review-only", "r8-fixture")
    metrics = summary["metrics"]
    assert len(summary["rows"]) == metrics["fixed_denominator"] == 9
    assert (metrics["attempted_rows"], metrics["not_run_rows"]) == (5, 4)
    assert metrics["raw_category_matches"] == 3
    assert metrics["valid_correct_first_causes"] == 2
    assert metrics["valid_ready_diagnoses"] == 2
    assert metrics["ready_valid_correct_first_causes"] == 1
    assert metrics["memory_targets"] == 1
    assert metrics["threshold_scope"] == "memory_targets_only"
    assert metrics["overall_acceptance"] == "not_assessed"
    assert summary["rows"][2]["first_round_diagnosis"]["raw_category_matches"] is None
    assert summary["rows"][4]["first_round_diagnosis"]["category"] == "optimizer_state"
    assert metrics["cost"]["actual_total_usd"] is None
    assert projection.amount(metrics["cost"]["known_actual_subtotal_usd"]) == projection.amount(".24")
    assert projection.amount(metrics["cost"]["charged_total_usd"]) == projection.amount(".30")
    assert metrics["cost"]["unknown_actual_runs"] == 1
    public = json.loads((tmp_path / "review-only/staging/published-evaluations/r8-fixture.json").read_text())
    assert len(public["rows"]) == 9 and "acceptance" not in public
    assert public["rows"][0]["peak_after_bytes"] is None
    assert public["rows"][1]["peak_after_bytes"] == 700
    assert public["rows"][1]["selected_patch_applied"] is True
    assert public["rows"][2]["first_cause_matches"] is None
    assert public["rows"][-1]["status"] == "not_run"
    assert public["rows"][-1]["provenance"] == "unknown"
    assert not any("private" in json.dumps(row) or "usage" in row for row in public["rows"])
    receipt = json.loads((tmp_path / "review-only/projection-receipt.json").read_text())
    assert receipt["script_sha256"] == hashlib.sha256(SCRIPT.read_bytes()).hexdigest()
    assert receipt["helper_source_sha256"] == projection.helper_hashes()
    assert receipt["published"] is False and receipt["provider_calls"] == receipt["gpu_jobs"] == 0
    assert original == {str(p): p.read_bytes() for p in campaign.parent.rglob("*") if p.is_file()}

@pytest.mark.parametrize("change", ["old_schema", "eight_rows", "duplicate_row", "wrong_revision", "other_scope"])
def test_old_results_changed_denominator_and_scope_refused(campaign, change):
    document = json.loads(campaign.read_text())
    if change == "old_schema":
        document["schema"] = "headroom.final48.repair.v1"
    elif change == "eight_rows":
        document["rows"].pop()
    elif change == "duplicate_row":
        document["rows"][1] = copy.deepcopy(document["rows"][0])
    elif change == "wrong_revision":
        document["revision"] = "r7"
    else:
        document["threshold_scope"] = "overall"
    write(campaign, document)
    with pytest.raises(ValueError):
        projection.collect(campaign)

def test_source_changed_after_read_and_symlink_are_refused(campaign):
    sources = projection.Sources(campaign.parent)
    sources.read("campaign.json")
    data = json.loads(campaign.read_text())
    data["revision"] = "r9"
    write(campaign, data)
    with pytest.raises(ValueError):
        sources.recheck()

    linked.symlink_to(campaign)

        sources.read("linked.json")


def test_existing_output_is_never_replaced(campaign, tmp_path):
    output = tmp_path / "existing-live-sentinel"
    output.mkdir()
    sentinel = write(output / "public-evaluations.json", {"do_not_overwrite": True})
    with pytest.raises(ValueError, match="new directory"):
        projection.stage_campaign(campaign, output, "r8-fixture")
    assert json.loads(sentinel.read_text()) == {"do_not_overwrite": True}


def test_helper_change_aborts_without_success_receipt(campaign, tmp_path, monkeypatch):
    calls = 0
    original = projection.helper_hashes()
    def changing_hashes():
        nonlocal calls
        calls += 1
        return original if calls == 1 else {**original, "changed": "yes"}
    monkeypatch.setattr(projection, "helper_hashes", changing_hashes)
    output = tmp_path / "incomplete-review"
    with pytest.raises(ValueError, match="helper source changed"):
        projection.stage_campaign(campaign, output, "r8-fixture")
    assert not (output / "projection-receipt.json").exists()
