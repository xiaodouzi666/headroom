class Latest(v.SupplementEvidence):
    def __init__(self):
        self.root = ROOT
        self.document = {**prev.document, "summary": {**prev.document["summary"], "latest_campaign": "R13"}}
        self.levels = copy.deepcopy(prev.levels)

    def selected(self, scenario):
        return {"scenario_id": scenario, "campaign": "R13"} if scenario == "s8_two_models_resident" else prev.selected(scenario)

    def value(self, name):
        return claims[name] if name in claims else prev.value(name)


evidence = Latest()
checked = []
errors = []
for name in ["README.md", "docs/DEVPOST.md", "docs/RESULTS.md"]:
    text = (ROOT / name).read_text()
    # RESULTS contains verbatim historical archives and unscoped historical tables.
    # Validate edited scoped lines here; explicit table/provenance checks follow.
    for number, line in enumerate(text.splitlines(), 1):
        if "<!-- numbers:" not in line:
            continue
        original = line
        normalized = re.sub(r"latest October 30 evaluation(?: \(R13\))?", "latest campaign", line)
        normalized = normalized.replace("best single evaluation", "best single campaign")
        if "numbers: summary" in line:
            rendered = v.rendered_line(line)
            bound = []
            for fraction in v.FRACTION.finditer(rendered):
                context = v.clause(rendered, fraction.start(), fraction.end())
                if "ready for patching" in context:
                    key = "R13.ready_for_patching"
                elif "threshold" in context:
                    key = "summary.minimum_targets"
                elif "raw first-category" in context:
                    key = "R13.raw_category_matches"
                elif "first-cause" in context:
                    assert "valid" in context and "correct" in context
                    key = "R13.valid_correct_first_causes"
                elif "best single evaluation" in context:
                    key = "summary.best_single_campaign_targets"
                elif "distinct scenario" in context and "histor" in context:
                    key = "summary.historical_target_scenario_count"
                elif "latest october 30 evaluation" in context:
                    key = "R13.targets"
                else:
                    raise AssertionError((name, number, context))
                assert int(fraction[1]) == claims[key] and int(fraction[2]) == 9
                bound.append({"claim": key, "value": claims[key], "denominator": 9})
            checked.append({"document": name, "line": number, "claims": bound, "text": original})
            continue
        try:
            result = v.verify_line(evidence, normalized)
            checked.append({"document": name, "line": number, "claims": result, "text": original})
        except Exception as error:
            errors.append({"document": name, "line": number, "error": str(error)})
