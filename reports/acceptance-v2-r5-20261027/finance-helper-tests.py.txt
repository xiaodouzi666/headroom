def mutate(data, key, change, *, repin=True):
    value = json.loads(data[key])
    change(value)
    data[key] = f.canonical(value)
    pin = {
        "before": "before_sha256",
        "after": "after_sha256",
        "probe_receipt": "probe_receipt_sha256",
        "base_profile_raw": "base_profile_sha256",
    }.get(key)
    if repin and pin:
        data[pin] = f.digest(data[key])

def fixture_r4_root():
    root = f.ROOT
    folder = root / "reports/acceptance-v2-r4-20261027"
    spec = importlib.util.spec_from_file_location(
        "r4_fixture_reader", folder / "test_financial_history_v2.py"
    )
    oldtests = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(oldtests)
    old = oldtests.fixture()
    r4_raw = {
        "audit": (folder / "independent-audit-history-v2-final.json").read_bytes(),
        "execution": (folder / "execution.json").read_bytes(),
        "cap": (folder / "cap-review.json").read_bytes(),
        "preflight": (folder / "preflight.json").read_bytes(),
        "campaign": (
            root / "runs/evaluation/live/20261027T002717226380Z-119f2528/campaign.json"
        ).read_bytes(),
    }
    r4_docs = {key: json.loads(value) for key, value in r4_raw.items()}
    funding, retirement, allocation = (
        json.loads((root / name).read_bytes())
        for name in (f.R4.FUNDING_PATH, f.R4.RETIREMENT_PATH, f.R4.ALLOCATION_PATH)
    )
    profile = json.loads((root / f.R4.PROFILE).read_bytes())
    hashes = {}
    for pin in f.R4.FUNDING_PINS:
        hashes[pin["path"]] = f.digest((root / pin["path"]).read_bytes())
        assert hashes[pin["path"]] == pin["sha256"]
    for entry in retirement["entrypoints"]:
        for key in ("path", "archived_path"):
            if key in entry:
                hashes[entry[key]] = f.digest((root / entry[key]).read_bytes())
    driver = f.R4.driver_funding_pins((folder / "run_mvp_v2.py").read_text())
    after = (root / ".cache/r5-low-effort-finance-20261027/ledger-before.json").read_bytes()
    assert f.digest(after) == "ce620d98b05d4532e5a27a93e1421b7f3be5caded08bccb3cf0092166a34d30f"
    return {
        "legacy_documents": old["documents"],
        "legacy_hashes": old["hashes"],
        "funding_args": [funding, retirement, allocation, profile, r4_docs["cap"], driver, hashes],
        "r4_documents": r4_docs,
        "r4_raw": r4_raw,
        "r4_after": after,
    }

        "new_row_drop",
        "duplicate",
        "audit",
        "funding",
        "legacy_review",
    ],
)

    data = fixture_r4_root()
    if mutation in {"old_amount", "old_row_drop", "new_row_drop", "duplicate"}:
        value = json.loads(data["r4_after"])
        if mutation == "old_amount":
            value["budget_attempts"][0]["charged_usd"] = "0"

            value["budget_attempts"].append(copy.deepcopy(value["budget_attempts"][0]))
        data["r4_after"] = f.canonical(value)
    elif mutation == "audit":
        data["r4_documents"]["audit"]["finance"]["all_time"]["conservative_charged_usd"] = "0"
    elif mutation == "funding":
        data["funding_args"][0]["runtime_conservative_net_usd"] = "0"
    else:
        data["legacy_documents"]["after"]["budget_attempts"][0]["review_sha256"] = "a" * 64
    with pytest.raises((ValueError, KeyError, TypeError)):
        f.verify_r4_root(**data)

def test_composition_proves_probe_start_and_cannot_authorize_dispatch(monkeypatch):
    # Root math has separate real pinned tests; this isolates orchestration order.
    data = fixture_probe()
    monkeypatch.setattr(
        f, "verify_r4_root", lambda **kwargs: {"accepted": True, "fixture_only": True}
    )
    pins = dict.fromkeys(
        (
            "r4_after",
            "probe_before",
            "probe_after",
            "probe_calls",
            "probe_receipt",
            "profile_transition",
            "new_profile",
            "r5_cap",
            "r5_preflight",
        ),
        "PENDING",
    )
    pins.update(
        r4_after=data["before_sha256"],
        probe_before=data["before_sha256"],
        probe_after=data["after_sha256"],
        probe_calls=f.digest(data["calls"]),
        probe_receipt=data["probe_receipt_sha256"],
    )
    result = f.verify_root_and_probe(
        r4_inputs={"r4_after": data["before"]}, probe_inputs=data, pins=pins
    )
    assert result["accepted"] and not result["dispatch_permitted"]
    assert not result["profile_transition_verified"]
    assert f.is_verified_probe_finance(result["probe_finance"])
    with pytest.raises(ValueError):
        f.require_frozen_pins(pins)  # Audit/dispatch cannot use probe-only readiness.
    bad_root = json.loads(data["before"])
    bad_root["budget_runs"][0]["day"] = "2026-10-25"
    raw = f.canonical(bad_root)
    pins["r4_after"] = f.digest(raw)
    with pytest.raises(ValueError, match="probe_begins_at_exact_r4_end"):
        f.verify_root_and_probe(r4_inputs={"r4_after": raw}, probe_inputs=data, pins=pins)

def test_request_anchor_is_independent_of_self_consistent_payload_and_report():
    data = fixture_probe()
    payload = json.loads(data["request_payload_raw"])
    payload["messages"][0]["content"] = "A different but fully self-consistent request."
    data["request_payload_raw"] = json.dumps(payload, ensure_ascii=False, allow_nan=False).encode()
    wire = json.dumps(payload, ensure_ascii=False, allow_nan=False, separators=(",", ":")).encode()
    fingerprint = f.digest(
        json.dumps(
            payload, sort_keys=True, ensure_ascii=False, allow_nan=False, separators=(",", ":")
        ).encode()
    )
    mutate(
        data,
        "calls",
        lambda x: x[0]["request_evidence"].update(
            wire_body_sha256=f.digest(wire),
            wire_body_utf8_bytes=len(wire),
            stored_payload_sha256=f.digest(data["request_payload_raw"]),
        ),
    )
    mutate(data, "probe_receipt", lambda x: x.__setitem__("request_sha256", fingerprint))
    with pytest.raises(ValueError, match="exact_wire_and_reviewed_request_fingerprint"):
        f.verify_probe_delta(**data)
    data["expected_request_sha256"] = "PENDING"
    with pytest.raises(ValueError, match="independent_expected_request_pin"):
        f.verify_probe_delta(**data)
