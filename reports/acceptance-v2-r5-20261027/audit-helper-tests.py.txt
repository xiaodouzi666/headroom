def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    obj = importlib.util.module_from_spec(spec)
    sys.modules[name] = obj
    spec.loader.exec_module(obj)
    return obj

def guard(event, args):
    if event.startswith("socket.") or event in {"sqlite3.connect", "subprocess.Popen", "os.system", "os.exec"}:
        raise AssertionError("Pure tests forbid network, SQLite and process execution")
    if event == "open" and isinstance(args[0], (str, bytes, os.PathLike)):
        p = Path(os.fsdecode(args[0]))
        if any(x.startswith(".env") for x in p.parts) or str(p) in {
            str(ROOT / ".cache/budget.db"), str(ROOT / "runs/headroom.db")
        }:
            raise AssertionError("Pure tests forbid credentials and production databases")


sys.addaudithook(guard)


def uid(value):
    return f"{value:032x}"


def run(value, limit="0.6"):
    return {"id": uid(value), "day": "2026-10-27", "limit_usd": limit,
            "status": "closed", "halt_reason": None, "budget_mode": "actual_usage"}


def attempt(value, run_id, amount="0.01", unknown=False):
    return {"id": uid(value), "run_id": run_id, "upper_usd": amount,
            "charged_usd": amount, "actual_usd": None if unknown else amount,
            "settled": 1, "admission_kind": "verified_provider_bound",
            "dispatched": 1, "charge_status": "reviewed_retained" if unknown else "actual",
            "charge_basis": "full_retention_review" if unknown else "provider_usage",
            "review_sha256": "b" * 64 if unknown else None}


def snapshots():
    prior = {
        "budget_runs": [run(1), run(2)],
        "budget_attempts": [
            attempt(101, uid(1), "0.0822432", True),
            attempt(102, uid(2), "0.0006072"),
        ],
        "budget_retention_reviews": [{
            "attempt_id": uid(101), "receipt_sha256": "b" * 64,
            "run_id": uid(1), "evidence_json": '{"full_retention":true}',
            "reviewed_at": "2026-10-27T00:00:00+00:00",
        }],
    }
    current = copy.deepcopy(prior)
    ids = [uid(i) for i in range(10, 19)]
    current["budget_runs"] += [run(i) for i in range(10, 19)]
    current["budget_attempts"] += [attempt(200 + i, uid(i)) for i in range(10, 19)]
    return prior, current, ids


def extension(prior, current, ids):
    return audit.r5_campaign_extension(
        finance, finance.canonical(prior), current["budget_runs"],
        current["budget_attempts"], current["budget_retention_reviews"], ids,
    )


def test_exact_addition_preserves_unknown_debt():
    prior, current, ids = snapshots()
    result = extension(prior, current, ids)
    assert result["accepted"] is True
    assert result["unknown_actual_never_zeroed"] is True
    assert prior["budget_attempts"][0]["actual_usd"] is None
    assert result["prior"]["conservative_charged_usd"] == "0.0828504"

@pytest.mark.parametrize("change", ["remove", "add", "evidence", "timestamp", "receipt"])
def test_retention_review_full_rows_fail(change):
    prior, current, ids = snapshots()
    if change == "remove":
        current["budget_retention_reviews"] = []
    elif change == "add":
        row = copy.deepcopy(current["budget_retention_reviews"][0])
        row["attempt_id"] = uid(999)
        current["budget_retention_reviews"].append(row)
    else:
        field = {"evidence": "evidence_json", "timestamp": "reviewed_at", "receipt": "receipt_sha256"}[change]
        current["budget_retention_reviews"][0][field] = "changed"
    # Ordinary ledger projection excludes retention review rows.
    assert finance.projection(finance.snapshot(finance.canonical(current))) == finance.projection(
        finance.snapshot(finance.canonical(snapshots()[1]))
    )
    assert extension(prior, current, ids)["accepted"] is False


@pytest.mark.parametrize("change", [
    "missing_run", "duplicate_run", "duplicate_attempt", "removed_history_attempt",
    "historical_reuse", "eight_ids", "ten_ids", "duplicate_ids", "non_uuid",
    "extra_foreign_run", "extra_call_on_old_run", "orphan_call",
    "case_limit", "case_mode",
])
def test_inventory_and_extra_calls_fail(change):
    prior, current, ids = snapshots()
    if change == "missing_run":
        current["budget_runs"].pop()
    elif change == "duplicate_run":
        current["budget_runs"].append(copy.deepcopy(current["budget_runs"][-1]))
    elif change == "duplicate_attempt":
        current["budget_attempts"].append(copy.deepcopy(current["budget_attempts"][-1]))
    elif change == "removed_history_attempt":
        current["budget_attempts"].pop(0)
    elif change == "historical_reuse":
        ids[-1] = uid(1)
    elif change == "eight_ids":
        ids.pop()
    elif change == "ten_ids":
        ids.append(uid(30))
    elif change == "duplicate_ids":
        ids[-1] = ids[0]
    elif change == "non_uuid":
        ids[-1] = "bad"
    elif change == "extra_foreign_run":
        current["budget_runs"].append(run(30))
    elif change == "extra_call_on_old_run":
        current["budget_attempts"].append(attempt(999, uid(1)))
    elif change == "orphan_call":
        current["budget_attempts"].append(attempt(999, uid(30)))
    elif change == "case_limit":
        current["budget_runs"][-1]["limit_usd"] = "0.7"
    elif change == "case_mode":
        current["budget_runs"][-1]["budget_mode"] = "reservations"
    assert extension(prior, current, ids)["accepted"] is False

def test_real_dispatch_positive_keeps_actual_unknown():
